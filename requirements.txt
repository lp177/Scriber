# A modern discord.py is required: 2.5.x fails Discord's current voice gateway
# handshake with close code 4006, so we need 2.7.x. discord.py[voice] 2.7 pulls
# in `davey` (required — discord.py refuses to connect to voice without it) and
# negotiates DAVE, Discord's voice end-to-end encryption, which servers can
# require (opting out is rejected with close code 4017). voice_recv predates
# DAVE and only strips transport encryption, so the still-E2EE payload decodes
# to garbage ("OpusError: corrupted stream"). scriber.bot.client bridges the two
# — it runs received audio through discord.py's davey MLS session so voice_recv
# gets plain Opus.
discord.py[voice]>=2.7,<3
# discord-ext-voice-recv only publishes pre-releases; pin the latest known-good
# alpha (naming a pre-release version makes pip accept it without --pre).
discord-ext-voice-recv==0.5.2a179
faster-whisper>=1.1
numpy>=1.26
fastapi>=0.115
python-multipart>=0.0.9
uvicorn>=0.30
httpx>=0.27
python-dotenv>=1.0
# Model Context Protocol SDK, used only by the optional MCP server
# (scriber/web/mcp_server.py). Scriber starts fine without it — the MCP
# listener is then skipped with a warning. >=1.9 for stateless streamable HTTP.
mcp>=1.9,<2
